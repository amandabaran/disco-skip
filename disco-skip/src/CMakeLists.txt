cmake_minimum_required(VERSION 3.9)
project(disco-skip CXX)

# ─── 1. Compile-time toggles (1 = enable, 0 = disable) ───────────────────────
#
# DS_CACHE_ENABLED / DS_N_REPLICAS are the two toggles mandated by
# docs/invariants.md §9.
#
# DS_N_REPLICAS DOES NOT SET THE REPLICA COUNT OF THIS BINARY. ds::kNumReplicas
# appears in exactly two places -- its own definition and a static_assert -- and
# nowhere in the operational path. The live replica count is conns_.size(), i.e.
# whatever `-s` says at run time, and majority() derives from that. So this
# staying at 1 while every experiment runs three servers is not a bug; the
# previous comment here claimed "at 1 the quorum is 1 and the single-replica
# path is taken", which is true of the TEST binaries (tests/Makefile builds
# repl1 and repl3 variants) and false of this one.
set(DS_CACHE_ENABLED 1)
set(DS_N_REPLICAS    1)

# Sizing of the local skip-vector cache. MAX_LAYERS is the compile-time level
# cap; IDX_EXP=3 gives a node capacity of 2<<3 = 16 entries, which must match
# the remote node capacity (interface doc A4).
set(DS_MAX_LAYERS 8)
set(DS_IDX_EXP    3)

# DATA node capacity, separate from the index fan-out above.
#
# 2<<DS_DATA_EXP entries per data vector: 16 / 32 / 64 / 128 at 3 / 4 / 5 / 6.
# Bigger data nodes cover more keys per node, so a range walk needs fewer of
# them and fewer round trips; the index fan-out (DS_IDX_EXP) governs descent
# depth and is a different question. These used to be one knob.
#
# VecRecord stays a multiple of 64 bytes at every value, with keys[] and vals[]
# both cache-line aligned -- see the layout note in ds_node.hpp.
set(DS_DATA_EXP   3)

# Maximum threads per client process. Currently one client thread per process,
# but this sizes the cache's hazard-pointer arrays, so it must be >= the real
# thread count (see include-cache/common/machine_defines.template.h).
set(DS_MAX_THREADS 4)

# Legacy flat-register (Chimera) ABD path. Kept compiling while the skip-vector
# Get/Insert are built out; removed once they replace it.
set(DS_REG_CACHE_ENABLED     1)
set(DS_REG_WRITEBACK_ENABLED 0)

if(NOT DS_N_REPLICAS EQUAL 1 AND NOT DS_N_REPLICAS EQUAL 3)
    message(FATAL_ERROR "DS_N_REPLICAS must be 1 or 3 (got ${DS_N_REPLICAS})")
endif()

# NO -mavx2. THESE CPUs DO NOT HAVE AVX2.
#
#   model name : Intel(R) Xeon(R) CPU E5-2450 0 @ 2.10GHz
#   flags      : ... avx ... sse4_2 ...        <-- no avx2
#
# E5-2450 (no "v2") is Sandy Bridge-EP. AVX2 arrived with Haswell in 2013, so
# an AVX2 instruction is an INVALID OPCODE here, and the kernel says so:
#
#   traps: disco-skip-exe[5165] trap invalid opcode ip:56e7f786cc61 ...
#
# 0 kops with the flag, 215 without. The faulting ip maps into lyra::opt --
# COMMAND-LINE PARSING -- because the compiler auto-vectorised early setup, so
# the process died before bootstrap while the log merely stopped after
# queue-pair setup. That misdirection cost a round of wrong theories: the first
# diagnosis here blamed under-aligned 32-byte moves, which was wrong. SIGILL,
# not SIGSEGV. An earlier version of this comment also asserted "all cluster
# nodes here are AVX2-capable", which was never checked and is false.
#
# IF SIMD IS REVISITED, THE CEILING IS SSE4.2, NOT AVX2. pcmpgtq gives TWO
# 64-bit compares per instruction, not four. Against a binary search over <=16
# keys that is worth 0.1-0.3% of a range operation (~176 comparisons per
# scan-100 against 63 us uncontended), so it is not worth a global codegen
# change. Confine it: __attribute__((target("sse4.2"))) on findLte alone plus a
# __builtin_cpu_supports dispatch.
#
# findLte's vector branch is guarded by __AVX2__, so with no flag here it
# compiles to the scalar path, while the tests still pass -mavx2 and bite-check
# the vector version on a build machine that has it.

add_compile_definitions(
    DS_CACHE_ENABLED=${DS_CACHE_ENABLED}
    DS_N_REPLICAS=${DS_N_REPLICAS}
    DS_MAX_LAYERS=${DS_MAX_LAYERS}
    DS_IDX_EXP=${DS_IDX_EXP}
    DS_DATA_EXP=${DS_DATA_EXP}
    DS_REG_CACHE_ENABLED=${DS_REG_CACHE_ENABLED}
    DS_REG_WRITEBACK_ENABLED=${DS_REG_WRITEBACK_ENABLED}
)

# ─── 2. Generate the cache's machine_defines.h ───────────────────────────────
#
# Only machine_defines.template.h is checked in (the real header is gitignored),
# so nothing that includes the cache compiles until we generate it. We write it
# into the build tree rather than the source tree so the source stays clean and
# the value tracks DS_MAX_THREADS above.
set(DS_CACHE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../include)
set(DS_GEN_DIR   ${CMAKE_CURRENT_BINARY_DIR}/gen/include-cache/common)
file(MAKE_DIRECTORY ${DS_GEN_DIR})
file(WRITE ${DS_GEN_DIR}/machine_defines.h
"#pragma once
// GENERATED by disco-skip/src/CMakeLists.txt -- do not edit.
// Derived from include-cache/common/machine_defines.template.h.
#define MAX_THREADS ${DS_MAX_THREADS}
#define NUMA_ZONES 1
#define THREADS_PER_CORE 1
#define TEMPLATE_INITIALIZATION 0
")

# ─── 3. Relax GCC 13 strict warning checks ───────────────────────────────────
#
# These stay WARNINGS -- we still get told -- they just do not fail the build.
# Each is here because the cache half triggers it and include/ is the other
# author's code.
if(CMAKE_COMPILER_IS_GNUCXX)
    add_compile_options(
        -Wno-error=switch-default
        -Wno-error=conversion
        -Wno-error=missing-declarations
        -Wno-error=undef
        # -Wstrict-overflow=5 cannot be handled the way the others are.
        #
        # The cache is full of signed descending loops -- `int layer = layers-1;
        # for (; layer >= 1; --layer)` and similar throughout
        # include/skipvector_disco.h -- which is exactly the `X - C1 cmp C2`
        # shape level 5 rejects. Marking the cache includes SYSTEM (below) does
        # not help: this diagnostic is emitted during optimisation, AFTER
        # inlining, and is attributed to whichever function absorbed the
        # arithmetic, which is ours. Its reported location is useless -- it
        # lands on a closing brace or on `int main()` -- so a scoped
        # `#pragma GCC diagnostic ignored` cannot be placed, and noinline on the
        # wrapper does not move it either. Both were tried.
        #
        # OUR OWN CODE IS STILL COVERED, because the local gate compiles the
        # DS_CACHE_ENABLED=0 arm with this fatal (tests/Makefile). That arm is
        # all of src/ with none of the cache, and it is what caught the two
        # real instances of this in our code: findLte's binary search and the
        # futures' `++counter > K` guards. So the check that matters is kept
        # where it can work, and demoted only where the cache is inlined in.
        -Wno-error=strict-overflow
    )
endif()

include(${CMAKE_BINARY_DIR}/setup.cmake)
dory_setup_cmake()

add_executable(disco-skip ${HEADER_TIDER} main.cpp)

# The cache half is header-only. Both generated-dir entries are needed: `gen`
# resolves the "include-cache/common/machine_defines.h" form used by the tests,
# and DS_GEN_DIR resolves the bare "machine_defines.h" that
# include-cache/common/config.h uses.
#
# SYSTEM, i.e. -isystem, is load bearing. The dory toolchain compiles with
# -Werror plus a much stricter warning set than the cache half is written for
# (its own include/Makefile uses only -Wall -Wextra -Wpedantic). Building it
# with plain -I fails on, among others:
#
#   -Werror=sign-promo      cpu_policy_tools.h, config.h -- an unsigned enum
#                           passed to std::to_string / operator<<
#   -Werror=class-memaccess vector_sfra.h -- memcpy over atomic_kv, which is
#                           deliberate and is exactly the relocation that
#                           ds_remote_addr.hpp's lock-free static_asserts
#                           exist to keep safe
#
# The cache is the other author's half of the repo and is not ours to edit, and
# blanket -Wno-error=... would weaken those checks on our own code too.
# -isystem suppresses diagnostics originating inside those headers while
# leaving full strictness everywhere else, which is precisely what it is for.
target_include_directories(disco-skip SYSTEM PRIVATE
    ${CMAKE_CURRENT_BINARY_DIR}/gen
    ${DS_GEN_DIR}
    ${DS_CACHE_DIR}
    ${DS_CACHE_DIR}/include-cache
)

target_link_libraries(disco-skip ${CONAN_LIBS})

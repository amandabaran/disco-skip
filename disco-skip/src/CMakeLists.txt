cmake_minimum_required(VERSION 3.9)
project(disco-skip CXX)

# ─── 1. Compile-time toggles (1 = enable, 0 = disable) ───────────────────────
#
# DS_CACHE_ENABLED / DS_N_REPLICAS are the two toggles mandated by
# docs/invariants.md §9. DS_N_REPLICAS must be 1 or 3: at 1 the quorum is 1 and
# the single-replica path is taken, which the microbench compares against 3.
set(DS_CACHE_ENABLED 1)
set(DS_N_REPLICAS    1)

# Sizing of the local skip-vector cache. MAX_LAYERS is the compile-time level
# cap; IDX_EXP=3 gives a node capacity of 2<<3 = 16 entries, which must match
# the remote node capacity (interface doc A4).
set(DS_MAX_LAYERS 8)
set(DS_IDX_EXP    3)

# Maximum threads per client process. Currently one client thread per process,
# but this sizes the cache's hazard-pointer arrays, so it must be >= the real
# thread count (see include-cache/common/machine_defines.template.h).
set(DS_MAX_THREADS 4)

# Legacy flat-register (Chimera) ABD path. Kept compiling while the skip-vector
# Get/Insert are built out; removed once they replace it.
set(DS_REG_CACHE_ENABLED     1)
set(DS_REG_WRITEBACK_ENABLED 0)

if(NOT DS_N_REPLICAS EQUAL 1 AND NOT DS_N_REPLICAS EQUAL 3)
    message(FATAL_ERROR "DS_N_REPLICAS must be 1 or 3 (got ${DS_N_REPLICAS})")
endif()

add_compile_definitions(
    DS_CACHE_ENABLED=${DS_CACHE_ENABLED}
    DS_N_REPLICAS=${DS_N_REPLICAS}
    DS_MAX_LAYERS=${DS_MAX_LAYERS}
    DS_IDX_EXP=${DS_IDX_EXP}
    DS_REG_CACHE_ENABLED=${DS_REG_CACHE_ENABLED}
    DS_REG_WRITEBACK_ENABLED=${DS_REG_WRITEBACK_ENABLED}
)

# ─── 2. Generate the cache's machine_defines.h ───────────────────────────────
#
# Only machine_defines.template.h is checked in (the real header is gitignored),
# so nothing that includes the cache compiles until we generate it. We write it
# into the build tree rather than the source tree so the source stays clean and
# the value tracks DS_MAX_THREADS above.
set(DS_CACHE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../include)
set(DS_GEN_DIR   ${CMAKE_CURRENT_BINARY_DIR}/gen/include-cache/common)
file(MAKE_DIRECTORY ${DS_GEN_DIR})
file(WRITE ${DS_GEN_DIR}/machine_defines.h
"#pragma once
// GENERATED by disco-skip/src/CMakeLists.txt -- do not edit.
// Derived from include-cache/common/machine_defines.template.h.
#define MAX_THREADS ${DS_MAX_THREADS}
#define NUMA_ZONES 1
#define THREADS_PER_CORE 1
#define TEMPLATE_INITIALIZATION 0
")

# ─── 3. Relax GCC 13 strict warning checks ───────────────────────────────────
if(CMAKE_COMPILER_IS_GNUCXX)
    add_compile_options(
        -Wno-error=switch-default
        -Wno-error=conversion
        -Wno-error=missing-declarations
        -Wno-error=undef
    )
endif()

include(${CMAKE_BINARY_DIR}/setup.cmake)
dory_setup_cmake()

add_executable(disco-skip ${HEADER_TIDER} main.cpp)

# The cache half is header-only. Both generated-dir entries are needed: `gen`
# resolves the "include-cache/common/machine_defines.h" form used by the tests,
# and DS_GEN_DIR resolves the bare "machine_defines.h" that
# include-cache/common/config.h uses.
target_include_directories(disco-skip PRIVATE
    ${CMAKE_CURRENT_BINARY_DIR}/gen
    ${DS_GEN_DIR}
    ${DS_CACHE_DIR}
    ${DS_CACHE_DIR}/include-cache
)

target_link_libraries(disco-skip ${CONAN_LIBS})
